#include <bits/stdc++.h>
using namespace std;

#define se second
#define pb push_back
#define fi first
#define el endl
#define ppi pair<pair<int, int>, int>

using ll = long long;
using db = double;
const int MOD = 1e9 + 7;
const ll INF = 1e18;

const int Maxn = 2e5 + 5;
int n, k, x[Maxn], visited[Maxn] = {0}; 

void Try(int i){
    for(int j = 1; j <= n; j++){
        if(visited[j] == 0){
            x[i] = j;
            visited[j] = 1;
        }
        else{
            continue;
        }
        if(i == n){
            for(int k = 1; k <= n ; k++) cout << x[k] << " ";
            cout << el;
        }
        else{
            Try(i + 1);
        }
        visited[j] = 0;
    }
}

int main(){
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    cin >> n;
    Try(1);

    return 0;
}
